from pathlib import Path import re, zipfile path = Path("/mnt/data/index.html") html = path.read_text(encoding="utf-8") # Replace the three logo SVG variants with a cleaner, upright, symmetrical mark. header_logo = '''''' large_logo = '''''' about_logo = '''''' # First SVG after brand opening html = re.sub( r'', header_logo, html, count=1, flags=re.S ) # Hero SVG html = re.sub( r'', large_logo, html, count=1, flags=re.S ) # About SVG: first remaining 240x240 SVG html = re.sub( r'', about_logo, html, count=1, flags=re.S ) # Ensure the hero mark itself cannot be visually rotated/skewed. html = html.replace( '.hero-tooth{position:relative;', '.hero-tooth{position:relative;transform:none;' ) path.write_text(html, encoding="utf-8") zip_path = Path("/mnt/data/naruse-dental-website-fixed-logo.zip") with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z: z.write(path, arcname="index.html") print("Logo sudah diluruskan dan file website versi terbaru dibuat.") print(zip_path)
